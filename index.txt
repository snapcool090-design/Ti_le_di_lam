// ===== SHIFT SELECTION FUNCTIONS =====

/**
 * Xác định ca mặc định dựa trên giờ hiện tại của client
 * Trước 15:00:00 -> Ca ngày, từ 15:00:01 -> Ca đêm
 */
function getDefaultShift() {
  const now = new Date();
  const hours = now.getHours();
  const minutes = now.getMinutes();
  const seconds = now.getSeconds();
  
  // Tính tổng giây từ 00:00:00
  const totalSeconds = hours * 3600 + minutes * 60 + seconds;
  
  // 15:00:00 = 15 * 3600 = 54000 giây
  return totalSeconds < 54000 ? 'day' : 'night';
}

/**
 * Khởi tạo shift selector với giá trị mặc định
 */
function initializeShiftSelector() {
  const shiftSelect = document.getElementById('stats-shift');
  if (shiftSelect) {
    const defaultShift = getDefaultShift();
    shiftSelect.value = defaultShift;
    console.log(`Auto-selected shift: ${defaultShift} (current time: ${new Date().toLocaleTimeString()})`);
  }
}

    // ===== LOGIN HANDLING =====
function handleLogin(event) {
  event.preventDefault();
  
  const genCode = document.getElementById('gen-code').value.trim();
  const loginBtn = document.getElementById('login-btn');
  const loginText = document.getElementById('login-text');
  const loginLoading = document.getElementById('login-loading');
  
  // Validate GEN code format
  if (!genCode || !/^\d{8}$/.test(genCode)) {
    showMessage('login-message', 'Vui lòng nhập mã GEN đúng 8 chữ số', 'error');
    return;
  }
  
  // Show loading
  loginText.textContent = 'Đang đăng nhập...';
  loginLoading.style.display = 'inline-block';
  loginBtn.disabled = true;
  
  google.script.run
    .withSuccessHandler(function(userData) {
      loginSuccess(userData);
    })
    .withFailureHandler(function(error) {
      loginFailed(error);
    })
    .checkGenCode(genCode);
}

// ---- Thay thế loginSuccess với logic kiểm tra mật khẩu admin ----
function loginSuccess(userData) {
  // userData is object returned from server via checkGenCode
  currentUser = userData;
  // store base user data but do not yet mark admin unlocked
  try {
    localStorage.setItem('user_data', JSON.stringify(userData));
  } catch (e) {
    console.warn('Could not persist user_data to localStorage', e);
  }
  // reset admin_unlocked to false by default on fresh login
  try {
    localStorage.setItem('admin_unlocked', 'false');
  } catch (e) {}

  // If role is admin -> require password modal before granting admin rights
  if (currentUser && currentUser.role === 'admin') {
    // show modal to request password; if user confirms and password ok -> unlock admin
    showAdminPasswordModal(currentUser.gen);
  } else {
    // Non-admin (leader/user): proceed normally
    finalizeLoginAndEnterApp(false); // not unlocked admin
  }

  // Reset login button state UI
  const loginBtn = document.getElementById('login-btn');
  const loginText = document.getElementById('login-text');
  const loginLoading = document.getElementById('login-loading');
  if (loginText) loginText.textContent = 'ĐĂNG NHẬP';
  if (loginLoading) loginLoading.style.display = 'none';
  if (loginBtn) loginBtn.disabled = false;
}

// ---- Hàm đóng / mở modal password admin (tạo DOM động) ----
function showAdminPasswordModal(genCode) {
  // ensure any existing modal removed
  const existing = document.getElementById('admin-password-modal');
  if (existing) existing.parentNode.removeChild(existing);

  // Build modal HTML (uses styles already in CSS)
  const modalHtml = `
    <div id="admin-password-modal" class="modal-overlay active" style="z-index:2000;">
      <div class="modal" style="max-width:420px;">
        <h3>🔒 Xác thực Admin</h3>
        <p>Vui lòng nhập mật khẩu:</p>
        <div style="margin-top:8px;">
          <input id="admin-password-input" type="password" class="form-control" placeholder="Nhập mật khẩu Admin" style="width:100%; box-sizing:border-box; margin-bottom:8px;">
          <div id="admin-password-error" style="color:#ea4335; margin-bottom:8px; display:none;"></div>
        </div>
        <div style="display:flex; gap:12px; justify-content:center; margin-top:12px;">
          <button id="admin-password-confirm" class="btn btn-success">Xác nhận</button>
          <button id="admin-password-skip" class="btn">Bỏ qua</button>
        </div>
      </div>
    </div>
  `;

  // append to body
  const tmp = document.createElement('div');
  tmp.innerHTML = modalHtml;
  document.body.appendChild(tmp.firstElementChild);

  // bind events
  const confirmBtn = document.getElementById('admin-password-confirm');
  const skipBtn = document.getElementById('admin-password-skip');
  const inputEl = document.getElementById('admin-password-input');
  const errorEl = document.getElementById('admin-password-error');
  const modalOverlay = document.getElementById('admin-password-modal');

  // clicking outside modal or pressing Escape => treat as Skip
  modalOverlay.addEventListener('click', function(e) {
    if (e.target === modalOverlay) {
      // user clicked overlay
      demoteToLeaderAndProceed('closed');
    }
  });

  document.addEventListener('keydown', function onEsc(e) {
    if (e.key === 'Escape') {
      // remove listener and treat as skip
      document.removeEventListener('keydown', onEsc);
      demoteToLeaderAndProceed('esc');
    }
  });

  confirmBtn.onclick = function() {
    const pw = (inputEl.value || '').toString();
    if (!pw) {
      errorEl.style.display = 'block';
      errorEl.textContent = 'Vui lòng nhập mật khẩu';
      return;
    }
    // disable confirm while checking
    confirmBtn.disabled = true;
    confirmBtn.textContent = 'Đang kiểm tra...';

    // call server to verify password
    google.script.run
      .withSuccessHandler(function(resp) {
        confirmBtn.disabled = false;
        confirmBtn.textContent = 'Xác nhận';
        // resp: { success: true, matched: bool, hasPassword: bool } or { success:false, error }
        if (!resp) {
          errorEl.style.display = 'block';
          errorEl.textContent = 'Lỗi máy chủ. Vui lòng thử lại.';
          return;
        }
        if (!resp.success) {
          // treat server error (e.g., gen not found) as demote
          errorEl.style.display = 'block';
          errorEl.textContent = 'Lỗi xác thực: ' + (resp.error || 'Không xác định') + '. Bạn sẽ vào với quyền Leader.';
          // small delay then demote
          setTimeout(function() { demoteToLeaderAndProceed('server_error'); }, 1200);
          return;
        }
        if (!resp.hasPassword) {
          // no password stored -> demote to leader
          errorEl.style.display = 'block';
          errorEl.textContent = 'Tài khoản không có mật khẩu. Chuyển thành Leader.';
          setTimeout(function() { demoteToLeaderAndProceed('no_password'); }, 900);
          return;
        }
        if (resp.matched) {
          // success -> unlock admin and proceed
          try {
            localStorage.setItem('admin_unlocked', 'true');
          } catch (e) {}
          // ensure currentUser.role remains 'admin'
          if (currentUser) currentUser.role = 'admin';
          closeAdminPasswordModal();
          finalizeLoginAndEnterApp(true);
        } else {
          errorEl.style.display = 'block';
          errorEl.textContent = 'Mật khẩu không đúng. Vui lòng thử lại hoặc chọn Bỏ qua để vào với quyền Leader.';
          // keep modal open for retry
        }
      })
      .withFailureHandler(function(err) {
        confirmBtn.disabled = false;
        confirmBtn.textContent = 'Xác nhận';
        errorEl.style.display = 'block';
        errorEl.textContent = 'Lỗi kết nối. Vui lòng thử lại hoặc Bỏ qua.';
      })
      .verifyAdminPassword(genCode, pw);
  };

  skipBtn.onclick = function() {
    demoteToLeaderAndProceed('skip_button');
  };

  // focus input
  setTimeout(function() {
    if (inputEl) inputEl.focus();
  }, 200);
}

function closeAdminPasswordModal() {
  const modal = document.getElementById('admin-password-modal');
  if (modal) {
    modal.classList.remove('active');
    try { modal.parentNode.removeChild(modal); } catch (e) {}
  }
}

// demote user to leader role and proceed into app (hide admin tab)
function demoteToLeaderAndProceed(reason) {
  // set admin_unlocked = false
  try {
    localStorage.setItem('admin_unlocked', 'false');
  } catch (e) {}
  // change currentUser.role to 'leader' if currentUser exists and had been admin
  if (currentUser && currentUser.role === 'admin') {
    currentUser.role = 'leader';
    try { localStorage.setItem('user_data', JSON.stringify(currentUser)); } catch (e) {}
  }
  // close modal if open
  closeAdminPasswordModal();
  // proceed into app but with reduced rights (finalize login)
  finalizeLoginAndEnterApp(false);
}

// finalize login and show main app; expects adminUnlocked boolean
function finalizeLoginAndEnterApp(adminUnlocked) {
  // if adminUnlocked is boolean true -> grant admin tab
  try {
    // if adminUnlocked === true then mark localStorage accordingly (already set earlier when matched)
    if (adminUnlocked) {
      try { localStorage.setItem('admin_unlocked', 'true'); } catch (e) {}

function showTab(tabName) {
  // Kiểm tra quyền truy cập tab
  if (!hasPermissionForTab(tabName)) {
    showMessage('survey-message', 'Bạn không có quyền truy cập tab này', 'error', 3000);
    return;
  }
  
  // Hide all tabs
  document.querySelectorAll('.tab-content').forEach(tab => {
    tab.classList.remove('active');
  });
  
  // Remove active class from all buttons
  document.querySelectorAll('.tab-btn').forEach(btn => {
    btn.classList.remove('active');
  });
  
  // Show selected tab
  const tabEl = document.getElementById(tabName);
  if (tabEl) tabEl.classList.add('active');
  
  // Activate selected button bằng data-tab
  const tabBtn = document.querySelector(`.tab-btn[data-tab="${tabName}"]`);
  if (tabBtn) tabBtn.classList.add('active');
  
  // Khởi tạo date picker khi chuyển tab (nếu cần)
  if (tabName === 'results' || tabName === 'admin') {
    setTimeout(() => {
      initializeDatePickers();
    }, 50);
  }
  if (tabName === 'admin') {
    setTimeout(() => {
      initializeAdminDatePicker();
    }, 50);
  }
  if (tabName === 'history') {
    setTimeout(() => {
      initializeHistoryDatePickers();
    }, 50);
  }
}

function hasPermissionForTab(tabName) {
  if (!currentUser || !currentUser.role) return false;
  
  const userRole = currentUser.role;
  
  switch(tabName) {
    case 'survey':
      return true; // Tất cả users đều được truy cập

    case 'results':
      return userRole === 'leader' || userRole === 'admin';

    case 'history':
      return true; // Tất cả users đều được truy cập

    case 'admin':
      return userRole === 'admin';

    default:
      return false;
  }
}

        'leader': 'Trưởng nhóm',

function showMessage(elementId, message, type, duration = 0) {
  const element = document.getElementById(elementId);
  if (!element) {
    // fallback: console + alert để người dùng thấy thông báo nếu phần tử không tồn tại
    console.warn('showMessage fallback:', elementId, message);
    alert(message);
    return;
  }
  
  element.innerHTML = message;
  element.className = `status-message status-${type}`;
  element.style.display = 'block';
  
  // CHỈ TỰ ẨN NẾU duration > 0
  if (duration > 0) {
    // Xóa bất kỳ timeout nào trước đó
    if (element.timeoutId) {
      clearTimeout(element.timeoutId);
    }
    
    // Tự động ẩn sau thời gian chỉ định
    element.timeoutId = setTimeout(() => {
      element.style.display = 'none';
      element.timeoutId = null;
    }, duration);
  } else {
    // duration = 0: KHÔNG TỰ ẨN, xóa timeout nếu có
    if (element.timeoutId) {
      clearTimeout(element.timeoutId);
      element.timeoutId = null;
    }
  }
}

// Hàm tạo message sau submit (giữ định dạng: mỗi dòng 1 div .msg-line)
function createDetailedSuccessMessage(userName, genCode, answers) {
  const lines = [];

  // Title (hiển thị nhưng không copy)
  lines.push('✅ Bạn đã hoàn thành khảo sát');

  // Name - GEN
  const nameGen = `${(userName || '').trim()}${userName ? ' - ' : ''}${(genCode || '').trim()}`.trim();
  if (nameGen) lines.push(nameGen);

  // Date (q2)
  const surveyDate = answers && answers.q2 && answers.q2.answer ? answers.q2.answer : '';
  if (surveyDate) lines.push(`Ngày: ${surveyDate}`);

  // Shift (q3)
  const shift = answers && answers.q3 && answers.q3.answer ? answers.q3.answer : '';
  if (shift) lines.push(shift);

  // Dynamic answers: q4, q5, ... (bỏ qua trống)
  const dynamicKeys = Object.keys(answers || {})
    .filter(k => /^q\d+$/.test(k))
    .map(k => ({ k, num: parseInt(k.replace('q', ''), 10) }))
    .filter(o => o.num >= 4)
    .sort((a, b) => a.num - b.num)
    .map(o => o.k);

  dynamicKeys.forEach(qId => {
    const obj = answers[qId];
    if (!obj) return;
    const ans = obj.answer != null ? String(obj.answer).trim() : '';
    const reason = obj.reason != null ? String(obj.reason).trim() : '';
    if (!ans) return;
    let line = ans;
    if (reason) line += ' - ' + reason;
    lines.push(line);
  });

  // Build HTML lines
  const htmlLines = lines.map((ln, idx) => {
    const cls = `msg-line msg-line-${idx}`;
    return `<div class="${cls}" style="margin-bottom:6px;">${escapeHtml(ln)}</div>`;
  }).join('');

  const html = `
    <div style="text-align:left;">
      ${htmlLines}
      <div style="margin-top:10px; display:flex; gap:8px; align-items:center; justify-content:center;">
        <button class="btn btn-primary" id="copy-survey-btn" onclick="copySurveyMessage()" style="padding:8px 12px; font-size:14px;">
          📋 Sao chép
        </button>
      </div>
    </div>
  `;

  return html;
}

// Escape HTML helper (nếu đã có, giữ nguyên)
function escapeHtml(text) {
  if (text == null) return '';
  return String(text)
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#039;');
}

// Copy function: bỏ dòng tiêu đề (dòng đầu .msg-line) khi copy
function copySurveyMessage() {
  const container = document.getElementById('survey-message');
  if (!container) {
    alert('Không tìm thấy vùng thông báo để copy.');
    return;
  }

  // Collect displayed .msg-line elements in order
  const lineEls = Array.from(container.querySelectorAll('.msg-line'));
  if (lineEls.length === 0) {
    alert('Không có nội dung để sao chép.');
    return;
  }

  // Skip first line (title) and collect the rest
  const lines = [];
  for (let i = 1; i < lineEls.length; i++) {
    const txt = (lineEls[i].innerText || lineEls[i].textContent || '').trim();
    if (txt) lines.push(txt);
  }

  const plain = lines.join('\n');

  if (!plain) {
    alert('Không có nội dung để sao chép.');
    return;
  }

  // Try navigator.clipboard first
  if (navigator && navigator.clipboard && navigator.clipboard.writeText) {
    navigator.clipboard.writeText(plain).then(function() {
      // Show inline confirmation if not already present
      if (!container.querySelector('.copy-note')) {
        const note = '<div class="copy-note" style="margin-top:8px;color:#155724;font-weight:600;text-align:center;">✅ Đã sao chép vào clipboard</div>';
        container.insertAdjacentHTML('beforeend', note);
      }
    }).catch(function(err) {
      fallbackCopyTextToClipboard(plain);
    });
  } else {
    fallbackCopyTextToClipboard(plain);
  }
}

// Fallback copy for older browsers
function fallbackCopyTextToClipboard(text) {
  try {
    const textarea = document.createElement('textarea');
    textarea.value = text;
    textarea.style.position = 'fixed';
    textarea.style.top = '-9999px';
    document.body.appendChild(textarea);
    textarea.focus();
    textarea.select();

    const successful = document.execCommand('copy');
    document.body.removeChild(textarea);

    const container = document.getElementById('survey-message');
    if (successful) {
      if (container && !container.querySelector('.copy-note')) {
        container.insertAdjacentHTML('beforeend', '<div class="copy-note" style="margin-top:8px;color:#155724;font-weight:600;text-align:center;">✅ Đã sao chép vào clipboard</div>');
      } else {
        alert('✅ Đã sao chép vào clipboard');
      }
    } else {
      alert('Không thể sao chép tự động. Vui lòng bôi đen và sao chép thủ công.');
    }
  } catch (e) {
    alert('Trình duyệt không hỗ trợ sao chép tự động. Vui lòng bôi đen và sao chép thủ công.');
  }
}

      // Chỉ gọi getEmployeeByGen khi currentUser tồn tại và có gen
      if (currentUser && currentUser.gen) {
        google.script.run
          .withSuccessHandler(function(respEmp) {
            let employeeLeader = '';
            if (respEmp && respEmp.success) {
              employeeLeader = respEmp.data.leaderName || '';
            }
            buildLeaderDropdown(leaders, employeeLeader);
          })
          .withFailureHandler(function(err) {
            console.error('Error getEmployeeByGen:', err);
            buildLeaderDropdown(leaders, '');
            showMessage('survey-message', 'Lỗi tải thông tin nhân viên', 'error');
          })
          .getEmployeeByGen(currentUser.gen);
      } else {
        // Không có currentUser => build dropdown chỉ với leaders, default rỗng
        buildLeaderDropdown(leaders, '');
      }
    })
    .withFailureHandler(function(err) {
      console.error('Error getLeadersList:', err);
      showMessage('survey-message', 'Lỗi kết nối khi tải danh sách leader', 'error');
    })
    .getLeadersList();
}

/**
 * Xây dropdown có search, chọn default theo employeeLeader nếu có,
 * nếu không có employeeLeader thì chọn phần tử đầu tiên (nếu tồn tại).
 */
function buildLeaderDropdown(leadersArray, employeeLeader = '') {
  const dropdown = document.getElementById('leader-dropdown');
  if (!dropdown) return;

  // 🔴 SỬA: Xử lý cả array và object response
  const leaders = Array.isArray(leadersArray) ? leadersArray : (leadersArray.data || []);
  
  // 🔴 SỬA: Ưu tiên employeeLeader từ backend, nếu không có thì dùng giá trị hiện tại
  const currentValue = employeeLeader || dropdown.value.trim(); 

  dropdown.innerHTML = '';

  leaders.forEach(name => {
    const trimmedName = name.trim();
    const opt = document.createElement('option');
    opt.value = trimmedName;
    opt.textContent = trimmedName;
    dropdown.appendChild(opt);
  });

  // 🔴 SỬA: Gán giá trị và SET BIẾN GLOBAL
  dropdown.value = currentValue;
  _originalLeaderForThisUser = currentValue; // 🔴 QUAN TRỌNG: Khởi tạo giá trị ban đầu

  dropdown.required = true;

  // Gắn sự kiện thay đổi
  dropdown.addEventListener('change', function () {
    const newVal = this.value;
    console.log('Leader changed:', _originalLeaderForThisUser, '→', newVal); // 🔴 DEBUG
    
    if (!newVal || newVal === _originalLeaderForThisUser) {
      _pendingLeaderSelection = '';
      return;
    }
    _pendingLeaderSelection = newVal;
    showConfirmLeaderModal(_originalLeaderForThisUser, _pendingLeaderSelection, dropdown);
  });
}




/* Hiển thị modal xác nhận và xử lý confirm/cancel (revert nếu cancel) */
function showConfirmLeaderModal(oldLeader, newLeader, dropdownElement) {
  const modal = document.getElementById('confirm-leader-modal');
  const msg = document.getElementById('confirm-leader-message');
  const yesBtn = document.getElementById('confirm-leader-yes');
  const noBtn = document.getElementById('confirm-leader-no');
  if (!modal || !msg || !yesBtn || !noBtn) return;

  msg.textContent = `Bạn có muốn cập nhật Leader của bạn từ "${oldLeader || '(trống)'}" thành "${newLeader}" không?`;
  modal.classList.add('active');

  yesBtn.onclick = null;
  noBtn.onclick = null;

  yesBtn.onclick = function() {
    yesBtn.disabled = true;
    noBtn.disabled = true;
    google.script.run
      .withSuccessHandler(function(resp) {
        yesBtn.disabled = false;
        noBtn.disabled = false;
        modal.classList.remove('active');

        if (resp && resp.success) {
          _originalLeaderForThisUser = newLeader;
          if (currentUser) {
            currentUser.leaderName = newLeader;
            try { localStorage.setItem('user_data', JSON.stringify(currentUser)); } catch(e){console.warn(e);}
          }
          showMessage('survey-message', 'Cập nhật leader thành công', 'success');
        } else {
          dropdownElement.value = _originalLeaderForThisUser || '';
          showMessage('survey-message', 'Cập nhật leader thất bại: ' + (resp && resp.error ? resp.error : 'Lỗi server'), 'error');
        }
        _pendingLeaderSelection = '';
      })
      .withFailureHandler(function(err) {
        yesBtn.disabled = false;
        noBtn.disabled = false;
        modal.classList.remove('active');
        dropdownElement.value = _originalLeaderForThisUser || '';
        _pendingLeaderSelection = '';
        showMessage('survey-message', 'Lỗi kết nối khi cập nhật leader: ' + err.message, 'error');
      })
      .updateEmployeeLeader(currentUser.gen, newLeader);
  };

  noBtn.onclick = function() {
    dropdownElement.value = _originalLeaderForThisUser || '';
    _pendingLeaderSelection = '';
    modal.classList.remove('active');
  };
}

// ===== REASON INPUT TOGGLE (for "Khác" option) =====
function toggleReasonInput(elem) {
  // elem is the radio input element that triggered onchange
  try {
    if (!elem) return;
    const questionName = elem.name; // e.g. 'q4' or 'q5'
    const selectedValue = elem.value;
    const placeholder = elem.dataset && elem.dataset.placeholder ? String(elem.dataset.placeholder) : '';

    // Determine option index from elem.id (expect format like q4_opt0)
    const idParts = (elem.id || '').split('_');
    const lastPart = idParts.length ? idParts[idParts.length - 1] : '';
    const optionIndex = lastPart.startsWith('opt') ? lastPart.replace('opt', '') : '';

    // For this question, hide all reason divs for other options and show only the matched one (if placeholder)
    const reasonDivPrefix = `reason-${questionName}-`;
    const reasonInputPrefix = `reason-input-${questionName}-`;

    // Find all elements that match prefix
    const allReasonDivs = Array.from(document.querySelectorAll(`[id^="${reasonDivPrefix}"]`));
    allReasonDivs.forEach(div => {
      // hide all initially
      div.style.display = 'none';
      // clear contained input if exists
      const input = div.querySelector('input');
      if (input) {
        input.required = false;
        // do not clear value here; only clear when hiding due to non-placeholder
        // but to be safe, clear when hiding
        input.value = '';
        input.placeholder = '';
      }
    });

    if (optionIndex === '') {
      // fallback: if no index, try find the reason div by questionName (single)
      const singleDiv = document.getElementById(`reason-${questionName}`);
      const singleInput = document.getElementById(`reason-input-${questionName}`);
      if (placeholder && placeholder.trim() !== '') {
        if (singleDiv) singleDiv.style.display = 'block';
        if (singleInput) {
          singleInput.style.display = 'block';
          singleInput.required = true;
          singleInput.placeholder = placeholder;
        }
      } else {
        if (singleDiv) singleDiv.style.display = 'none';
        if (singleInput) {
          singleInput.style.display = 'none';
          singleInput.required = false;
          singleInput.value = '';
          singleInput.placeholder = '';
        }
      }
      return;
    }

    // Show the specific reason div for selected option if placeholder present
    const targetReasonDiv = document.getElementById(`${reasonDivPrefix}${optionIndex}`);
    const targetReasonInput = document.getElementById(`${reasonInputPrefix}${optionIndex}`);

    if (placeholder && placeholder.trim() !== '') {
      if (targetReasonDiv) targetReasonDiv.style.display = 'block';
      if (targetReasonInput) {
        targetReasonInput.style.display = 'block';
        targetReasonInput.required = true;
        targetReasonInput.placeholder = placeholder;
      }
    } else {
      if (targetReasonDiv) targetReasonDiv.style.display = 'none';
      if (targetReasonInput) {
        targetReasonInput.style.display = 'none';
        targetReasonInput.required = false;
        targetReasonInput.value = '';
        targetReasonInput.placeholder = '';
      }
    }
  } catch (e) {
    console.error('toggleReasonInput error:', e);
  }
}

/**
 * Hiển thị/ẩn date picker khi chọn option "Ngày khác"

function toggleCustomDateInput() {
  const customOption = document.getElementById('survey-date-custom');
  const dateContainer = document.getElementById('custom-date-container');
  const dateInput = document.getElementById('custom-date-input');
  const errorDiv = document.getElementById('custom-date-error');

  if (!customOption || !dateContainer || !dateInput) return;

    dateContainer.style.display = 'block';

    setTimeout(() => {
      if (!dateInput.value) {
        const tomorrow = new Date();
        tomorrow.setDate(tomorrow.getDate() + 1);
        dateInput.value = formatDateForPicker(tomorrow);
      }
      dateInput.focus();
    }, 50);
    
    // Ẩn thông báo lỗi nếu có
    if (errorDiv) {
      errorDiv.style.display = 'none';
      errorDiv.textContent = '';
    }
  } else {
    // Ẩn date picker và clear value
    dateContainer.style.display = 'none';
    dateInput.value = '';
    
    // Ẩn thông báo lỗi
    if (errorDiv) {
      errorDiv.style.display = 'none';
      errorDiv.textContent = '';
    }
  }
}

/**
 * Khởi tạo custom date input với ngày mặc định là ngày mai
 */
function initializeCustomDateInput() {
  const dateInput = document.getElementById('custom-date-input');
  if (!dateInput) return;

  // Set min date là hôm nay (không cho chọn ngày quá khứ)
  const today = new Date();
  dateInput.min = formatDateForPicker(today);

  // Set default value là ngày mai
  const tomorrow = new Date(today);
  tomorrow.setDate(tomorrow.getDate() + 2);
  dateInput.value = formatDateForPicker(tomorrow);
}
/**
 * Validate ngày được chọn không được là quá khứ
 * Gọi khi thay đổi radio button hoặc thay đổi date picker
 */
function validateCustomDate() {
  const dateInput = document.getElementById('custom-date-input');
  const errorDiv = document.getElementById('custom-date-error');
  const customOption = document.getElementById('survey-date-custom');

  if (!dateInput || !errorDiv) return;

  // Chỉ validate và hiển thị lỗi nếu option "Ngày khác" được chọn
  const isCustomSelected = customOption && customOption.checked;
  
  if (!dateInput.value) {
    if (isCustomSelected) {
      errorDiv.textContent = 'Vui lòng chọn ngày';
      errorDiv.style.display = 'block';
    } else {
      errorDiv.style.display = 'none';
    }
    return false;
  }

  const selectedDate = new Date(dateInput.value);
  const today = new Date();
  today.setHours(0, 0, 0, 0); // Reset về 00:00:00

  if (selectedDate < today) {
    if (isCustomSelected) {
      errorDiv.textContent = 'Không được chọn ngày trong quá khứ';
      errorDiv.style.display = 'block';
    }
    return false;
  } else {
    errorDiv.style.display = 'none';
    errorDiv.textContent = '';
    return true;
  }
}

// Hàm format ngày thành "dd/mm/yyyy"
function formatDate(date) {
  const day = String(date.getDate()).padStart(2, '0');
  const month = String(date.getMonth() + 1).padStart(2, '0');
  const year = date.getFullYear();
  return `${day}/${month}/${year}`;
}

function getWeekdayName(date) {
  const weekdays = ['Chủ nhật', 'Thứ hai', 'Thứ ba', 'Thứ tư', 'Thứ năm', 'Thứ sáu', 'Thứ bảy'];
  return weekdays[date.getDay()];
}

// ===== COMPREHENSIVE FORM VALIDATION =====
function validateSurveyForm() {
  const questionCards = document.querySelectorAll('.question-card');
  const errors = [];

  questionCards.forEach(card => {
    if (card.style.display !== 'none') {
      let questionId = 'không xác định';
      let hasAnswer = false;

      // 1. Kiểm tra radio - ĐẶC BIỆT XỬ LÝ CÂU 2
      const radioInputs = card.querySelectorAll('input[type="radio"][required]');
      if (radioInputs.length > 0) {
        questionId = radioInputs[0].name || questionId;
        
        // Đặc biệt xử lý Câu 2 (q2) có option "Ngày khác"
        if (questionId === 'q2') {
          const q2Today = document.getElementById('survey-date-today');
          const q2Tomorrow = document.getElementById('survey-date-tomorrow');
          const q2Custom = document.getElementById('survey-date-custom');
          const customDateInput = document.getElementById('custom-date-input');
          
          hasAnswer = (q2Today && q2Today.checked) || 
                      (q2Tomorrow && q2Tomorrow.checked) || 
                      (q2Custom && q2Custom.checked);
          
          // Nếu chọn "Ngày khác", kiểm tra có nhập date không
          if (q2Custom && q2Custom.checked) {
            if (!customDateInput || !customDateInput.value) {
              errors.push('Vui lòng chọn ngày cho option "Ngày khác"');
            } else if (!validateCustomDate()) {
              errors.push('Ngày không hợp lệ: không được chọn ngày trong quá khứ');
            }
          }
        } else {
          // Các câu radio khác
          hasAnswer = Array.from(radioInputs).some(input => input.checked);
        }

        if (!hasAnswer) {
          errors.push(`Bạn chưa trả lời câu hỏi ${questionId}`);
        }

        // Nếu chọn "Khác", kiểm tra lý do (cho các câu khác q2)
        if (questionId !== 'q2') {
          const selected = Array.from(radioInputs).find(input => input.checked);
          if (selected && selected.value === 'Khác') {
            const reasonInput = card.querySelector('.reason-input');
            if (!reasonInput || reasonInput.value.trim() === '') {
              errors.push(`Vui lòng nhập lý do cho câu hỏi ${questionId}`);
            }
          }
        }
        return; // đã xử lý radio, không cần kiểm tra tiếp
      }

      // 2. Kiểm tra select
      const selectInput = card.querySelector('select[required]');
      if (selectInput) {
        questionId = selectInput.id || questionId;
        if (!selectInput.value || selectInput.value.trim() === '') {
          errors.push(`Bạn chưa chọn đáp án cho câu hỏi ${questionId}`);
        }
        return;
      }

      // 3. Kiểm tra input text
      const textInput = card.querySelector('input[type="text"][required]');
      if (textInput) {
        questionId = textInput.id || questionId;
        if (!textInput.value || textInput.value.trim() === '') {
          errors.push(`Bạn chưa nhập thông tin cho câu hỏi ${questionId}`);
        }
        return;
      }
    }
  });

  return {
    isValid: errors.length === 0,
    errors: errors
  };
}

          html += `<td><span class="clickable-number" onclick="showEmployeeDetailedModal('${escapeHtml(leaderName)}', '${leader.employeeType}', '${key}', '${statistics.filterDate || ''}', '${safeLabel}')">${formatNumber(count)}</span></td>`;
        });
      });

      html += `</tr>`;
    });
  });

  html += `</tbody></table>`;
  container.innerHTML = html;
}

function displayEmployeeDetailedList(data) {
  const content = document.getElementById('employee-detailed-content');
  const copyBtn = document.getElementById('modal-copy-btn');

  if (!content) return;
  
  content.innerHTML = '';

  // Determine mode by answerType
  const answerType = data.answerType || '';
  const employees = Array.isArray(data.employees) ? data.employees.slice() : [];

  // Normalize timestamp string to Date for sorting; keep original string for display
  employees.forEach(emp => {
    emp._tsDate = null;
    if (emp.timestamp) {
      const d = new Date(emp.timestamp);
      if (!isNaN(d.getTime())) emp._tsDate = d;
      else {
        const parsed = parseDateString(emp.timestamp);
        if (parsed) emp._tsDate = parsed;
      }
    }
  });

  // Sorting rules
  if (answerType === 'pending' || answerType === 'total') {
    // Kiểu 1: Pending/Total - sắp xếp theo tên A-Z
    employees.sort((a, b) => {
      const na = (a.name || '').toLowerCase();
      const nb = (b.name || '').toLowerCase();
      if (na < nb) return -1;
      if (na > nb) return 1;
      return 0;
    });
  } else {
    // Kiểu 2: On Leave + Other Answers - sắp xếp theo thời gian (cũ → mới)
    employees.sort((a, b) => {
      const ta = a._tsDate ? a._tsDate.getTime() : 0;
      const tb = b._tsDate ? b._tsDate.getTime() : 0;
      return ta - tb;
    });
  }

  // Build table structure - LUÔN 6 CỘT CHO TẤT CẢ (TRỪ PENDING)
  let tableHtml = `
      <table class="stats-table" style="width:100%; white-space: nowrap;">
        <thead>
          <tr>
  `;

  if (answerType === 'pending' || answerType === 'total') {
    // 🔥 KIỂU 1: PENDING/TOTAL (3 cột)
    tableHtml += `
            <th style="text-align: center; width: 60px;">STT</th>
            <th style="text-align: center; width: 100px;">GEN</th>
            <th style="text-align: center;">Họ Tên</th>
          </tr>
        </thead>
        <tbody>
    `;
    employees.forEach((emp, idx) => {
      tableHtml += `
        <tr>
          <td style="text-align: center;">${idx + 1}</td>
          <td style="text-align: center;">${escapeHtml(emp.gen)}</td>
          <td style="text-align: left;">${escapeHtml(emp.name)}</td>
        </tr>
      `;
    });
  } else {
    // 🔥 KIỂU 2: TẤT CẢ CÁC LOẠI KHÁC (6 cột)
    // Đổi tên cột "Loại nghỉ" thành "Đáp án" cho rõ ràng hơn
    tableHtml += `
            <th style="text-align: center; width: 60px;">STT</th>
            <th style="text-align: center; width: 100px;">GEN</th>
            <th style="text-align: center;">Họ Tên</th>
            <th style="text-align: center;">Lựa chọn</th>
            <th style="text-align: center;">Lý do</th>
            <th style="text-align: center; width: 120px;">Thời gian</th>
          </tr>
        </thead>
        <tbody>
    `;
    employees.forEach((emp, idx) => {
      const ts = emp._tsDate ? formatTimestampModal(emp._tsDate) : '';
      // Sử dụng emp.leaveType cho "Đáp án" (đã được set trong getEmployeesDetailedList)
      tableHtml += `
        <tr>
          <td style="text-align: center;">${idx + 1}</td>
          <td style="text-align: center;">${escapeHtml(emp.gen)}</td>
          <td style="text-align: left;">${escapeHtml(emp.name)}</td>
          <td style="text-align: left;">${escapeHtml(emp.leaveType || '')}</td>
          <td style="text-align: left;">${escapeHtml(emp.reason || '')}</td>
          <td style="text-align: center;">${escapeHtml(ts)}</td>
        </tr>
      `;
    });
  }

  tableHtml += `</tbody></table>`;

  content.innerHTML = tableHtml;

  if (copyBtn) {
    copyBtn.disabled = employees.length === 0;
  }
}

// Helper: format timestamp to dd/mm HH:mm (bỏ năm)
function formatTimestampModal(date) {
  if (!date) return '';
  const dd = String(date.getDate()).padStart(2, '0');
  const mm = String(date.getMonth() + 1).padStart(2, '0');
  const hh = String(date.getHours()).padStart(2, '0');
  const min = String(date.getMinutes()).padStart(2, '0');
  return `${dd}/${mm} ${hh}:${min}`;
}

// Thay thế hàm copyModalContent hiện tại bằng hàm dưới đây
function copyModalContent() {
  const headersContainer = document.getElementById('employee-detailed-headers');
  const contentContainer = document.getElementById('employee-detailed-content');

  if (!contentContainer) {
    console.warn('copyModalContent: employee-detailed-content not found');
    return;
  }

  try {
    // Lấy context từ dataset
    const answerType = headersContainer && headersContainer.dataset && headersContainer.dataset.answerType ? headersContainer.dataset.answerType : '';
    const modalTitle = headersContainer && headersContainer.dataset && headersContainer.dataset.modalTitle ? headersContainer.dataset.modalTitle : '';
    const surveyDate = headersContainer && headersContainer.dataset && headersContainer.dataset.surveyDate ? headersContainer.dataset.surveyDate : '';
    const leader = headersContainer && headersContainer.dataset && headersContainer.dataset.leader ? headersContainer.dataset.leader : '';

    // Parse table rows
    const table = contentContainer.querySelector('table');
    if (!table) {
      showModalCopyNote('Không có dữ liệu để sao chép');
      return;
    }

    const rows = Array.from(table.querySelectorAll('tbody tr'));
    if (rows.length === 0) {
      showModalCopyNote('Không có dữ liệu để sao chép');
      return;
    }

    const lines = [];

    // 🔴 THÊM HEADER VÀO COPY
    if (modalTitle) lines.push(modalTitle);
    if (surveyDate) lines.push(`Ngày: ${surveyDate}`);
    if (leader) lines.push(`Leader: ${leader}`);
    lines.push('Danh sách chi tiết:');

    // Xử lý dữ liệu
    if (answerType === 'pending' || answerType === 'total') {
      // Format Pending/Total: GEN[TAB]Họ Tên
      rows.forEach(tr => {
        const cells = tr.querySelectorAll('td');
        if (cells.length >= 3) {
          const gen = (cells[1].innerText || '').trim();
          const name = (cells[2].innerText || '').trim();
          if (gen) lines.push(`${gen}\t${name}`);
        }
      });
    } else {
      // Format cho tất cả các loại khác: GEN[TAB]Họ Tên[TAB]Đáp án[TAB]Lý do
      rows.forEach(tr => {
        const cells = tr.querySelectorAll('td');
        if (cells.length >= 6) {
          const gen = (cells[1].innerText || '').trim();
          const name = (cells[2].innerText || '').trim();
          const answer = (cells[3].innerText || '').trim();  // Cột "Đáp án"
          const reason = (cells[4].innerText || '').trim();  // Cột "Lý do"
          if (gen) lines.push(`${gen}\t${name}\t${answer}\t${reason}`);
        } else if (cells.length >= 3) {
          // Fallback
          const gen = (cells[1].innerText || '').trim();
          const name = (cells[2].innerText || '').trim();
          if (gen) lines.push(`${gen}\t${name}`);
        }
      });
    }

    const plain = lines.join('\n').trim();
    if (!plain || lines.length <= 4) { // Chỉ có header, không có dữ liệu
      showModalCopyNote('Không có nội dung để sao chép');
      return;
    }

    // Thực hiện copy
    if (navigator && navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(plain).then(function() {
        showModalCopyNote('✅ Đã sao chép vào clipboard');
      }).catch(function() {
        fallbackCopyTextToClipboardModal(plain);
      });
    } else {
      fallbackCopyTextToClipboardModal(plain);
    }

  } catch (e) {
    console.error('copyModalContent error', e);
    showModalCopyNote('Lỗi khi sao chép nội dung');
  }
}


// Keep or add the modal fallback copy helper (no alert)
function fallbackCopyTextToClipboardModal(text) {
  try {
    const textarea = document.createElement('textarea');
    textarea.value = text;
    textarea.style.position = 'fixed';
    textarea.style.top = '-9999px';
    document.body.appendChild(textarea);
    textarea.focus();
    textarea.select();

    const successful = document.execCommand && document.execCommand('copy');
    document.body.removeChild(textarea);

    if (successful) {
      showModalCopyNote('✅ Đã sao chép vào clipboard');
    } else {
      showModalCopyNote('Không thể sao chép tự động. Vui lòng bôi đen và sao chép thủ công.');
    }
  } catch (e) {
    showModalCopyNote('Trình duyệt không hỗ trợ sao chép tự động. Vui lòng bôi đen và sao chép thủ công.');
  }
}

// Helper hiển thị note trong modal (giữ nguyên nếu bạn đã có)
function showModalCopyNote(message) {
  try {
    // 1) Ưu tiên hiển thị ở vùng ngay dưới 2 nút hành động (nếu tồn tại)
    const actionNoteContainer = document.getElementById('employee-detailed-action-note');

    // Helper tạo/ghi đè note element
    function renderNote(container) {
      if (!container) return;

      // Nếu đã có note trước đó, tái sử dụng node
      let note = container.querySelector('.modal-copy-note');
      if (!note) {
        note = document.createElement('div');
        note.className = 'modal-copy-note';
        note.style.marginTop = '8px';
        note.style.color = '#155724';
        note.style.fontWeight = '600';
        note.style.textAlign = 'center';
        container.appendChild(note);
      }
      note.textContent = message;

      // Quản lý timeout để tự ẩn sau 4.5s
      if (note._timeoutId) {
        clearTimeout(note._timeoutId);
        note._timeoutId = null;
      }
      note._timeoutId = setTimeout(() => {
        try {
          if (note && note.parentNode) note.parentNode.removeChild(note);
        } catch (e) {}
        note._timeoutId = null;
      }, 4500);
    }

    if (actionNoteContainer) {
      renderNote(actionNoteContainer);
      return;
    }

    // 2) Fallback: nếu không tìm thấy actionNoteContainer, sử dụng vùng content (cách cũ)
    const content = document.getElementById('employee-detailed-content');
    if (!content) return;

    let note = content.querySelector('.modal-copy-note');
    if (!note) {
      note = document.createElement('div');
      note.className = 'modal-copy-note';
      note.style.marginTop = '12px';
      note.style.color = '#155724';
      note.style.fontWeight = '600';
      note.style.textAlign = 'center';
      content.appendChild(note);
    }

    note.textContent = message;

    if (note._timeoutId) {
      clearTimeout(note._timeoutId);
      note._timeoutId = null;
    }

    note._timeoutId = setTimeout(() => {
      try {
        if (note && note.parentNode) note.parentNode.removeChild(note);
      } catch (e) {}
      note._timeoutId = null;
    }, 4500);

  } catch (e) {
    console.error('showModalCopyNote error:', e);
  }
}

function collectSurveyAnswers() {
  const answers = {};

  // LẤY LEADER (Câu 1)
  const leaderDropdown = document.getElementById('leader-dropdown');
  const leader = leaderDropdown ? leaderDropdown.value : '';

  // LẤY CÂU 2: Xử lý đặc biệt vì có option "Ngày khác"
  const q2Today = document.getElementById('survey-date-today');
  const q2Tomorrow = document.getElementById('survey-date-tomorrow');
  const q2Custom = document.getElementById('survey-date-custom');
  const customDateInput = document.getElementById('custom-date-input');

  let q2Answer = '';
  if (q2Today && q2Today.checked) {
    q2Answer = q2Today.value; // dd/mm/yyyy
  } else if (q2Tomorrow && q2Tomorrow.checked) {
    q2Answer = q2Tomorrow.value; // dd/mm/yyyy
  } else if (q2Custom && q2Custom.checked && customDateInput && customDateInput.value) {
    // Convert từ yyyy-mm-dd sang dd/mm/yyyy
    const dateParts = customDateInput.value.split('-');
    if (dateParts.length === 3) {
      q2Answer = `${dateParts[2]}/${dateParts[1]}/${dateParts[0]}`;
    }
  }

  if (q2Answer) {
    answers['q2'] = {
      answer: q2Answer,
      reason: ''
    };
  }

  // LẤY TẤT CẢ CÂU TRẢ LỜI TỪ RADIO BUTTONS (đã checked) - TRỪ CÂU 2
  const questionInputs = document.querySelectorAll('input[type="radio"]:checked');
  
  questionInputs.forEach(input => {
    const questionId = input.name; // e.g., 'q3', 'q4'
    
    // Bỏ qua câu 2 vì đã xử lý ở trên
    if (questionId === 'q2') return;
    
    const answer = input.value;

    answers[questionId] = {
      answer: answer,
      reason: ''
    };

    // Thu thập lý do: tìm option index từ input.id (format q4_opt0)
    const idParts = (input.id || '').split('_');
    const lastPart = idParts.length ? idParts[idParts.length - 1] : '';
    let optionIndex = '';
    if (lastPart && lastPart.startsWith('opt')) {
      optionIndex = lastPart.replace('opt', '');
    }

    // Compose reason input id (new format reason-input-qX-<index>)
    if (optionIndex !== '') {
      const reasonInputId = `reason-input-${questionId}-${optionIndex}`;
      const reasonInput = document.getElementById(reasonInputId);
      if (reasonInput) {
        answers[questionId].reason = reasonInput.value.trim();
      }
    } else {
      // Fallback: older single reason-input-qX
      const fallback = document.getElementById(`reason-input-${questionId}`);
      if (fallback) {
        answers[questionId].reason = fallback.value.trim();
      }
    }
  });

  // LẤY THÔNG TIN USER TỪ currentUser
  const userName = currentUser ? currentUser.name : '';
  const employeeType = currentUser ? currentUser.employeeType : '';

  console.log('✅ Final collected data:', { leader, answers, userName, employeeType });

  return {
    leader: leader,
    answers: answers,
    userName: userName,
    employeeType: employeeType
  };
}

function shouldCollectAnswer(questionId, q1Answer) {
  // Luôn thu thập Câu 1
  if (questionId === 'q1') return true;
  
  // Nếu không có đáp án Câu 1, không thu thập gì cả
  if (!q1Answer) return false;
  
  // Chỉ thu thập Câu 2,3 nếu Câu 1 KHÔNG phải là Nghỉ nhóm ca
  if (questionId === 'q2' || questionId === 'q3') {
    return q1Answer.value !== 'Nghỉ nhóm ca';
  }
  
  // Thu thập các câu khác
  return true;
}

// Hàm chuyển đổi Date object sang định dạng YYYY-MM-DD cho input type="date"
function formatDateForPicker(date) {
  const year = date.getFullYear();
  const month = String(date.getMonth() + 1).padStart(2, '0');
  const day = String(date.getDate()).padStart(2, '0');
  return `${year}-${month}-${day}`;
}

// Hàm chuyển đổi từ YYYY-MM-DD sang dd/mm/yyyy
function formatDateFromPicker(dateString) {
  if (!dateString) return '';
  
  const parts = dateString.split('-');
  if (parts.length === 3) {
    return `${parts[2]}/${parts[1]}/${parts[0]}`;
  }
  return dateString;
}

function copyAttendanceReport(shift) {
  const fullText = document.getElementById('attendance-report-content').innerText;
  let content = '';
  
  // Lấy tiêu đề và ngày từ phần đầu
  const titleIndex = fullText.indexOf('[CNC PQC 3P]');
  const dateIndex = fullText.indexOf('Ngày:');
  const firstShiftIndex = fullText.indexOf('Ca ngày');
  
  // Lấy phần header (tiêu đề và ngày)
  let header = '';
  if (titleIndex !== -1 && firstShiftIndex !== -1) {
    header = fullText.substring(titleIndex, firstShiftIndex).trim();
  }
  
  if (shift === 'day') {
    // Copy ca ngày: lấy từ "Ca ngày" đến trước "Ca đêm"
    const dayStart = fullText.indexOf('Ca ngày');
    const nightStart = fullText.indexOf('Ca đêm');
    
    if (dayStart !== -1 && nightStart !== -1) {
      const dayContent = fullText.substring(dayStart, nightStart).trim();
      content = header + '\n\n' + dayContent;
    }
  } else {
    // Copy ca đêm: chỉ lấy phần "Ca đêm" đến hết
    const nightStart = fullText.indexOf('Ca đêm');
    
    if (nightStart !== -1) {
      const nightContent = fullText.substring(nightStart).trim();
      content = header + '\n\n' + nightContent;
    }
  }
  
  const messageText = `✅ Đã sao chép báo cáo ca ${shift === 'day' ? 'ngày' : 'đêm'}`;
  copyTextWithNotification(content, 'attendance-copy-message', messageText);
}

// Hàm copy chung với thông báo inline (giống tab Khảo sát)
function copyTextWithNotification(text, messageElementId, successMessage) {
  // Thử dùng clipboard API hiện đại trước
  if (navigator.clipboard && navigator.clipboard.writeText) {
    navigator.clipboard.writeText(text).then(() => {
      showInlineCopyMessage(messageElementId, successMessage);
    }).catch(() => {
      fallbackCopyTextWithNotification(text, messageElementId, successMessage);
    });
  } else {
    fallbackCopyTextWithNotification(text, messageElementId, successMessage);
  }
}

// Fallback copy với thông báo inline
function fallbackCopyTextWithNotification(text, messageElementId, successMessage) {
  try {
    const textarea = document.createElement('textarea');
    textarea.value = text;
    textarea.style.position = 'fixed';
    textarea.style.top = '-9999px';
    document.body.appendChild(textarea);
    textarea.focus();
    textarea.select();

    const successful = document.execCommand('copy');
    document.body.removeChild(textarea);

    if (successful) {
      showInlineCopyMessage(messageElementId, successMessage);
    } else {
      showInlineCopyMessage(messageElementId, '❌ Không thể sao chép tự động');
    }
  } catch (err) {
    showInlineCopyMessage(messageElementId, '❌ Trình duyệt không hỗ trợ sao chép');
  }
}

/**
 * Khởi tạo date picker cho tab Tra Cứu với giá trị mặc định:
 * - Nếu ngày hiện tại <= 5: A = 01/tháng trước, B = ngày cuối/tháng trước
 * - Nếu ngày hiện tại > 5: A = 01/tháng hiện tại, B = ngày cuối/tháng hiện tại
 */
function initializeHistoryDatePickers() {
  const now = new Date();
  const day = now.getDate();

  let fromDate, toDate;

  if (day <= 5) {
    // Tháng trước
    fromDate = new Date(now.getFullYear(), now.getMonth() - 1, 1);
    toDate = new Date(now.getFullYear(), now.getMonth(), 0); // Ngày cuối của tháng trước
  } else {
    // Tháng hiện tại
    fromDate = new Date(now.getFullYear(), now.getMonth(), 1);
    toDate = new Date(now.getFullYear(), now.getMonth() + 1, 0); // Ngày cuối của tháng hiện tại
  }

  const fromInput = document.getElementById('history-date-from');
  const toInput = document.getElementById('history-date-to');

  if (fromInput) fromInput.value = formatDateForPicker(fromDate);
  if (toInput) toInput.value = formatDateForPicker(toDate);

  console.log('History date pickers initialized: from=' + formatDateForPicker(fromDate) + ', to=' + formatDateForPicker(toDate));
}

 * Helper: download CSV content as file
 */
function downloadCSV(csvContent, filename) {
  try {
    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
    const link = document.createElement('a');
    if (link.download !== undefined) {
      const url = URL.createObjectURL(blob);
      link.setAttribute('href', url);
      link.setAttribute('download', filename);
      link.style.visibility = 'hidden';
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);
      URL.revokeObjectURL(url);
    } else {
      // Fallback for very old browsers
      const encodedUri = 'data:text/csv;charset=utf-8,' + encodeURIComponent(csvContent);
      window.open(encodedUri);
    }
  } catch (e) {
    console.error('downloadCSV error:', e);
    alert('Lỗi khi tải file: ' + e.message);
  }
}

/**
 * Export all employees (Admin only)
 * Calls backend exportAllEmployeesCSV() and downloads the CSV file
 */
function exportAllEmployees(button) {
  const btnText = button.querySelector('.btn-text');
  const btnLoading = button.querySelector('.btn-loading');
  const msgDiv = document.getElementById('export-employees-message');

  // Set loading state
  button.disabled = true;
  if (btnText) btnText.textContent = 'Đang xuất...';
  if (btnLoading) btnLoading.style.display = 'inline-block';

  google.script.run
    .withSuccessHandler(function(response) {
      // Reset button
      button.disabled = false;
      if (btnText) btnText.textContent = '📥 Export Excel';
      if (btnLoading) btnLoading.style.display = 'none';

      if (response && response.success) {
        downloadCSV(response.csv, response.filename);
        if (msgDiv) {
          msgDiv.innerHTML = '✅ Đã xuất ' + response.count + ' nhân viên';
          msgDiv.className = 'status-message status-success';
          msgDiv.style.display = 'block';
          setTimeout(function() { msgDiv.style.display = 'none'; }, 4000);
        }
      } else {
        if (msgDiv) {
          msgDiv.innerHTML = '❌ Lỗi: ' + escapeHtml((response && response.error) || 'Không xác định');
          msgDiv.className = 'status-message status-error';
          msgDiv.style.display = 'block';
        }
      }
    })
    .withFailureHandler(function(error) {
      // Reset button
      button.disabled = false;
      if (btnText) btnText.textContent = '📥 Export Excel';
      if (btnLoading) btnLoading.style.display = 'none';

      if (msgDiv) {
        msgDiv.innerHTML = '❌ Lỗi kết nối: ' + escapeHtml(error.message || String(error));
        msgDiv.className = 'status-message status-error';
        msgDiv.style.display = 'block';
      }
    })
    .exportAllEmployeesCSV();
}
